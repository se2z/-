cmake_minimum_required(VERSION 3.25)
project(notify_bot LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "notify_bot requires Windows and MSVC")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "notify_bot requires an x64 toolchain; configure with -A x64")
endif()
enable_testing()
add_library(notify_core STATIC
  src/config.cpp src/json.cpp src/hwid.cpp src/license.cpp src/formatting.cpp
  src/proxy_preflight.cpp src/send_guard.cpp src/scheduler.cpp src/winhttp_client.cpp src/gui.cpp)
target_include_directories(notify_core PUBLIC include)
target_compile_features(notify_core PUBLIC cxx_std_20)
target_compile_definitions(notify_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(notify_core PUBLIC /W4 /permissive- /utf-8)
target_link_libraries(notify_core PUBLIC winhttp.lib ws2_32.lib bcrypt.lib gdi32.lib msimg32.lib user32.lib advapi32.lib)
add_executable(notify_bot WIN32 src/main_gui.cpp)
target_link_libraries(notify_bot PRIVATE notify_core)
target_link_options(notify_bot PRIVATE /SUBSYSTEM:WINDOWS)
add_executable(notify_cli src/cli.cpp)
target_link_libraries(notify_cli PRIVATE notify_core)
add_executable(notify_tests tests/selftest_main.cpp tests/test_pure.cpp)
target_link_libraries(notify_tests PRIVATE notify_core)
add_test(NAME pure COMMAND notify_tests)
set_tests_properties(pure PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
